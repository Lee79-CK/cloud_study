Terraform & Ansible 강의 · Level 5 통합·MSP 실전 · 24편 / 26편 · 시리즈 목차

요구사항 한 장에서 드리프트 리포트까지 종합 프로젝트 — 요구사항 → Terraform 모듈 조립 → Ansible 역할 구성(Molecule) → CI에서 plan 리뷰·apply → 구성 검증·드리프트 리포트, 전체 코드 트리와 체크리스트

1~23편에서 만든 조각 — label·vpc·alb·app-servers·rds 모듈, 테스트와 정책, 기반 역할과 nginx_site, 동적 인벤토리, Vault, EE, AWX, 통합 패턴 — 을 가상의 고객 custa의 쇼핑몰 dev 환경 하나로 조립합니다. 그리고 그 저장소를 CI가 하듯 처음부터 끝까지 돌립니다: 정적 검사, plan과 정책, apply, 동적 인벤토리로 서버 구성, 2회차 멱등성, 일부러 만든 드리프트의 리포트까지. AWS 대신 AWS API 모의 서버(moto)에 실제로 40개 자원을 apply했고, 그 결과를 aws_ec2 플러그인이 찾아 2편 랩 노드에 구성을 적용했습니다.

module "label" {
  source = "../../../modules/label"

  project   = var.project
  env       = var.env
  component = "network"
  owner     = var.owner
}

# 2) 네트워크 모듈: label 모듈의 출력을 입력으로 받는다 (모듈 조합)
module "vpc" {
  source = "../../../modules/vpc"

  name       = module.label.name
  cidr_block = var.vpc_cidr
  az_count   = 2
  tags       = module.label.tags

  enable_nat_gateway   = false # 실습은 끈다. 켜면 시간당 과금
  enable_ssm_endpoints = var.enable_ssm_endpoints
}

# 3) 다른 state가 읽을 수 있도록 공개할 값만 SSM 파라미터로 내보낸다 (5편 12장)
resource "aws_ssm_parameter" "vpc_id" {
  name  = "/${var.project}/${var.env}/network/vpc_id"
  type  = "String"
  value = module.vpc.vpc_id
  tags  = module.label.tags
}

resource "aws_ssm_parameter" "private_subnet_ids" {
  name  = "/${var.project}/${var.env}/network/private_subnet_ids"
  type  = "StringList"
  value = join(",", values(module.vpc.private_subnet_ids))
  tags  = module.label.tags
}

# 7편에서 추가: ALB가 쓸 퍼블릭 서브넷도 공개한다
resource "aws_ssm_parameter" "public_subnet_ids" {
  name  = "/${var.project}/${var.env}/network/public_subnet_ids"

  source = "../../../modules/rds-postgres"

  name       = "${module.label.name}-db"
  vpc_id     = local.vpc_id
  subnet_ids = local.private_subnet_ids

  # 키 "app"은 고정 문자열 → for_each 키가 plan 시점에 확정된다

  backup_retention_period = 1

  # 실습 모드: 모듈의 안전한 기본값을 명시적으로 완화
  deletion_protection = !var.lab_mode
  skip_final_snapshot = var.lab_mode
  apply_immediately   = var.lab_mode
}
live/dev/app/variables.tf — admin_cidrs
variable "admin_cidrs" {
  description = "ALB에 접근할 수 있는 대역 (사무실·VPN). 실행하는 곳의 IP를 자동으로 넣지 않는다"
  type        = list(string)

  validation {
    condition     = length(var.admin_cidrs) > 0 && alltrue([for c in var.admin_cidrs : can(cidrhost(c, 0)) && c != "0.0.0.0/0"])
    error_message = "admin_cidrs 는 1개 이상의 CIDR 이어야 하고 0.0.0.0/0 은 허용하지 않습니다."
  }
}

# 6편 네트워크 state가 공개한 값만 읽는다 (state 파일 접근 권한 불필요, 5편 12장)
# String·StringList 파라미터는 비밀이 아니므로 insecure_value로 읽어 sensitive 전파를 막는다
data "aws_ssm_parameter" "vpc_id" {
  name = "/${var.project}/${var.env}/network/vpc_id"
}

data "aws_ssm_parameter" "public_subnet_ids" {
  name = "/${var.project}/${var.env}/network/public_subnet_ids"
}

data "aws_ssm_parameter" "private_subnet_ids" {
  name = "/${var.project}/${var.env}/network/private_subnet_ids"
}


locals {
  vpc_id             = data.aws_ssm_parameter.vpc_id.insecure_value
  public_subnet_ids  = split(",", data.aws_ssm_parameter.public_subnet_ids.insecure_value)
  private_subnet_ids = split(",", data.aws_ssm_parameter.private_subnet_ids.insecure_value)
}
7편의 checkip를 지운 이유

7편은 실습자의 PC에서 apply하므로 data "http"로 “내 IP”를 조회해 ALB 접근을 열었습니다. 같은 코드를 CI에서 돌리면 GitHub 러너의 IP가 들어가 매번 다른 IP로 plan이 바뀌고(드리프트 잡음), 고객 사무실은 막히고, 아무 러너 IP나 열립니다. 파이프라인으로 옮기는 순간 “실행하는 곳에 따라 결과가 달라지는 코드”는 모두 변수로 바꿔야 합니다. 대신 검증(validation)으로 0.0.0.0/0을 막았습니다.

# nginx_site 역할 테스트 — 이미 있는 시험 서버를 쓰는 방식 (컨테이너 드라이버 없이)
driver:
  name: default
  options:
    managed: false           # 만들고 지우는 것은 Molecule 이 하지 않는다
platforms:
  - name: molecule-node

  name: ansible
  inventory:
    hosts:
      all:
        hosts:
          molecule-node:
            # CI 에서는 시험용 서버 주소·키를 환경 변수로 (기본값은 2편 랩 node3)
            ansible_host: ${MOLECULE_HOST:-127.0.0.1}
            ansible_port: ${MOLECULE_PORT:-2223}
            ansible_user: ${MOLECULE_USER:-ansible}
            ansible_ssh_private_key_file: ${MOLECULE_KEY:-/root/iac-lab/lab/keys/lab_ed25519}
            ansible_ssh_common_args: "-o StrictHostKeyChecking=no"
verifier:
  name: ansible
scenario:
  test_sequence:
    - syntax
    - converge

    - cron: "0 23 * * *"   # 매일 08:00 KST
  workflow_dispatch:

permissions:
  id-token: write
  contents: read
  issues: write

jobs:
  drift:
    runs-on: ubuntu-24.04
    environment: dev-readonly
    steps:
      - uses: actions/checkout@v7
      - uses: aws-actions/configure-aws-credentials@v6
        with:
          role-to-assume: ${{ vars.TF_PLAN_ROLE_ARN }}
          aws-region: ap-northeast-2
      - uses: hashicorp/setup-terraform@v4
        with:
          terraform_version: 1.16.0
          terraform_wrapper: false

        id: tf
        working-directory: terraform/live/dev/app
        run: |
          terraform init -input=false -backend-config=backend/dev.s3.tfbackend
          set +e
          terraform plan -input=false -var-file=dev.tfvars -detailed-exitcode -out=drift.tfplan
          code=$?
          set -e
          [ "$code" = "1" ] && exit 1
          terraform show -json drift.tfplan > drift.json
          echo "code=$code" >> "$GITHUB_OUTPUT"
      - name: 서버 구성 드리프트 (AWX 점검 잡 결과 출력)
        env:
          CONTROLLER_HOST: ${{ vars.AWX_HOST }}
          CONTROLLER_OAUTH_TOKEN: ${{ secrets.AWX_TOKEN }}

          pip install --quiet awxkit==24.6.1 "setuptools<81"
          job=$(awx job_templates launch "${{ vars.AWX_CHECK_JT_ID }}" --monitor -f json | jq -r .id)
          awx jobs stdout "$job" > ansible_check.log
      - name: 리포트 만들기
        run: python3 tools/drift_report.py terraform/live/dev/app/drift.json ansible_check.log > drift_report.md
      - name: 드리프트가 있으면 이슈 생성
        if: steps.tf.outputs.code == '2' || hashFiles('drift_report.md') != ''
        uses: actions/github-script@v9
        with:
          script: |
            const fs = require('fs');
            const body = fs.readFileSync('drift_report.md', 'utf8');
            if (!body.includes('| ') || (body.includes('총 0건') && body.match(/총 0건/g).length === 2)) { return; }
            await github.rest.issues.create({
              owner: context.repo.owner, repo: context.repo.repo,
              title: `[drift] custa-shop dev ${new Date().toISOString().slice(0, 10)}`,
              body, labels: ['drift'],
            });

# 드리프트 리포트: Terraform plan JSON + Ansible check 출력 → Markdown
import json, sys, re
tf = json.load(open(sys.argv[1])); ans = open(sys.argv[2]).read()
print("## 드리프트 리포트 — custa-shop dev\n")
print("### 1. 클라우드 자원 (Terraform plan)\n")
rows = []
for rc in tf.get("resource_changes", []):
    acts = rc["change"]["actions"]
    if acts == ["no-op"] or acts == ["read"]: continue
    b, a = rc["change"].get("before") or {}, rc["change"].get("after") or {}
    keys = sorted(k for k in set(b) | set(a) if b.get(k) != a.get(k) and k not in ("tags_all",))
    detail = []
    for k in keys:
        if k == "tags":
            for tk in sorted(set(b.get(k) or {}) | set(a.get(k) or {})):
                if (b.get(k) or {}).get(tk) != (a.get(k) or {}).get(tk):
                    detail.append(f"tags.{tk}: {(b.get(k) or {}).get(tk)!r} → {(a.get(k) or {}).get(tk)!r}")
        else:
            detail.append(f"{k}: {b.get(k)!r} → {a.get(k)!r}")
    rows.append((rc["address"], "/".join(acts), "; ".join(detail)[:160]))
print("| 리소스 | 동작 | 차이 |\n|---|---|---|")
for r in rows: print(f"| `{r[0]}` | {r[1]} | {r[2]} |")
print(f"\n총 {len(rows)}건\n")
print("### 2. 서버 구성 (Ansible --check --diff)\n")
cur = None; items = []
for line in ans.splitlines():
    m = re.match(r"TASK \[(.+?)\]", line)
    if m: cur = m.group(1)
    m2 = re.match(r"changed: \[(.+?)\]", line)
    if m2 and cur: items.append((m2.group(1), cur))
print("| 서버 | 바뀔 태스크 |\n|---|---|")
for h, tname in items: print(f"| {h} | {tname} |")
print(f"\n총 {len(items)}건")

13과제

과제 1. prd 환경 추가

과제 2. 드리프트 리포트 개선

drift_report.py를 고쳐 ① 핸들러 예약은 별도 절로 ② “알려진 기준선 잡음” 목록(파일)과 일치하는 항목은 접어서 ③ 드리프트가 0건이면 “이상 없음” 한 줄만 출력하게 하세요. drift.yml의 이슈 생성 조건도 이에 맞게 단순화합니다.

과제 3. 실제 dev 계정에서 한 바퀴

여러분의 AWS 실습 계정(2편)에서 이 저장소를 apply하고(SSM 엔드포인트 또는 NAT 필요, 17편 12장), inventories/dev로 site.yml을 실행해 보세요. moto에서 나온 보안 그룹 규칙 잡음 3건이 실제 AWS에서도 나오는지 확인하고, 끝나면 반드시 destroy합니다.

과제 4. Molecule 시나리오 늘리기

base_ssh 역할에 Molecule 시나리오를 추가하세요. verify에서 sshd -T로 실제 적용 값을 검사하고, 잘못된 값(존재하지 않는 옵션)을 넣었을 때 validate가 파일을 쓰지 않는지 확인합니다. 스스로 잠기지 않도록 시험 서버 접속 경로를 어떻게 지킬지도 설계합니다.

과제 5. 요구사항 추가 → 끝까지

새 요구사항 “REQ-10: 웹 서버 로그를 중앙 로그 서버로 보낸다”를 추가하고, 표 → group_vars(base_logging_remote) → 보안 그룹 아웃바운드(Terraform) → PR → 적용 → 검증 태스크까지 한 번에 진행해 보세요. 어떤 파일들이 함께 바뀌는지 정리하면 23편 책임 경계표의 실제 모습이 됩니다.
