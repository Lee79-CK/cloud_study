Game 뷰 60 FPS
position = (0.00, 0.00) · 경과 시간 0.0초 · 누적 프레임 0
드래그 · 방향키 · WASD로 조작합니다. 캔버스를 클릭해야 키보드 입력이 들어갑니다.
Assets/Scripts/Player.cs
using UnityEngine;

public class Player : MonoBehaviour
{
    // 인스펙터에서 값을 조절할 수 있게 노출
    [SerializeField] private float speed = 5f;

    // 매 프레임 자동 호출 (초당 약 60회)
    void Update()
    {
        float h = Input.GetAxis("Horizontal"); // 좌우: -1 ~ 1
        float v = Input.GetAxis("Vertical");   // 상하: -1 ~ 1

        Vector3 dir = new Vector3(h, v, 0f);

        // deltaTime을 곱해야 기기 성능과 무관하게 같은 속도
        transform.position += dir * speed * Time.deltaTime;
    }
}
왼쪽 데모와 이 코드의 관계

속도 슬라이더 = speed 값, 드래그/방향키 = Input, 상자의 좌표 = transform.position. 유니티에서 여러분이 하는 일의 90%는 "어떤 값을, 언제, 얼마나 바꿀지" 정하는 것입니다.

01 배경 지식

엔진이 대신 해주는 일을 알아야
내가 할 일이 보인다

유니티는 "게임을 만들어 주는 프로그램"이 아니라, 게임에 공통으로 필요한 지겨운 작업을 미리 처리해 둔 거대한 도구 상자입니다. 무엇이 이미 들어 있는지 알면 바퀴를 다시 발명하지 않게 됩니다.

게임 엔진이 없으면 직접 만들어야 하는 것들

화면에 삼각형 하나를 그리려면 원래 그래픽 API(OpenGL/Vulkan/Metal)를 직접 다루고, 셰이더를 작성하고, 기기별 화면 크기를 계산해야 합니다. 엔진은 이걸 전부 감싸서 transform.position = ... 한 줄로 바꿔 줍니다.

기능 영역엔진이 없으면유니티에서는
렌더링그래픽 API 직접 호출, 셰이더 작성, 드로우콜 관리Sprite Renderer / Mesh Renderer 컴포넌트를 붙인다
물리충돌 판정·중력·마찰 수식을 직접 구현Rigidbody + Collider를 붙인다
입력안드로이드/iOS 터치 이벤트를 각각 처리Input / Input System이 통합해 준다
사운드오디오 버퍼·믹싱을 직접 관리AudioSource 컴포넌트
빌드플랫폼별 빌드 스크립트 작성Build Settings에서 플랫폼 전환
에셋 관리이미지/사운드 포맷 변환 파이프라인 구축파일을 Project 창에 끌어다 놓으면 자동 임포트
신입이 가장 많이 하는 오해

"엔진을 배우는 것 = 게임 개발을 배우는 것"이 아닙니다. 엔진은 도구이고, 실제 실력은 게임 로직 설계와 문제를 작은 단위로 쪼개는 능력에서 나옵니다. 버튼 위치를 외우는 데 시간을 쓰지 말고, 작은 게임을 끝까지 완성하는 경험에 시간을 쓰세요.

설치: 버전 선택이 첫 번째 의사결정

유니티는 Unity Hub라는 관리 프로그램을 먼저 설치하고, 그 안에서 에디터 버전을 골라 설치합니다. 버전은 크게 세 종류입니다.

Tech Stream / 프리뷰

새 기능을 먼저 넣은 버전. 학습·실험용으로는 좋지만 출시 프로젝트에 쓰면 버그를 직접 감당해야 합니다.

구버전

튜토리얼을 따라가려고 2019 같은 옛 버전을 깔지 마세요. 지금은 지원이 끝났고 최신 안드로이드 요구사항을 못 맞춥니다.

모바일 개발에 반드시 체크해야 하는 설치 모듈

  • Documentation — 오프라인 매뉴얼. 용량이 여유 있으면 같이 설치하면 편합니다.

유니티를 지탱하는 4개의 단어

이 네 가지만 정확히 이해하면 유니티의 구조는 끝입니다. 나머지는 전부 응용입니다.

1. GameObject — 빈 그릇

씬에 존재하는 모든 것. 캐릭터, 카메라, 조명, 버튼, 심지어 눈에 안 보이는 매니저까지 전부 GameObject입니다. 그 자체로는 아무 기능이 없습니다. 이름표가 달린 빈 상자라고 생각하세요.

2. Component — 붙이는 기능

빈 상자에 붙이는 능력. Sprite Renderer를 붙이면 보이고, Rigidbody2D를 붙이면 떨어지고, 내가 만든 Player.cs를 붙이면 움직입니다. 기능을 상속이 아니라 조립으로 만드는 방식입니다.

3. Scene — 무대 단위

오브젝트들의 배치를 저장한 파일. 보통 Title, MainGame, Result처럼 화면 단위로 나눕니다. 씬을 바꾸면 이전 씬의 오브젝트는 기본적으로 모두 파괴됩니다.

4. Prefab — 오브젝트 붕어빵 틀

완성된 GameObject를 파일로 저장한 것. 적 100마리를 손으로 배치하는 대신, 적 프리팹 하나를 만들고 코드로 Instantiate 합니다. 프리팹을 수정하면 모든 사본에 반영됩니다.

GameObject와 Component의 관계 (개념 코드)
// 유니티 내부를 아주 단순화하면 이런 구조입니다
class GameObject
{
    public string name;             // "Player"
    public Transform transform;     // 위치·회전·크기 (항상 존재, 제거 불가)
    public List<Component> components;  // 붙어 있는 기능들
}

// 적 프리팹을 화면에 5마리 생성하기
[SerializeField] private GameObject enemyPrefab;

void SpawnEnemies()
{
    for (int i = 0; i < 5; i++)
    {
        Vector3 pos = new Vector3(i * 2f, 0f, 0f);
        Instantiate(enemyPrefab, pos, Quaternion.identity);
    }
}

좌표계와 단위 — 숫자의 의미를 정하는 약속

유니티는 왼손 좌표계를 쓰며, X는 오른쪽, Y는 위, Z는 화면 안쪽입니다. 2D 게임에서는 Z를 거의 쓰지 않고 앞뒤 정렬(Sorting Order)로 처리합니다. 그리고 1 unit = 1 미터로 취급하는 것이 표준입니다. 물리 엔진의 중력값(-9.81)이 미터 기준으로 맞춰져 있기 때문에, 캐릭터를 100 unit 크기로 만들면 물리가 달나라처럼 느리게 보입니다.

Transform 실습 슬라이더를 움직여 보세요
지금 이 상태를 코드로 쓰면
transform.position = new Vector3(0.0f, 0.0f, 0f);
회전은 도(degree) 단위이고, 화면의 회색 격자 한 칸이 1 unit입니다.

2D와 3D — 무엇이 실제로 다른가

둘은 완전히 다른 엔진이 아니라 같은 엔진의 다른 설정입니다. 차이는 기본 카메라 투영 방식과 사용하는 물리 모듈입니다.

2D 프로젝트3D 프로젝트
카메라Orthographic(원근 없음)Perspective(원근 있음)
물리Rigidbody2D, Collider2D, Physics2DRigidbody, Collider, Physics
그래픽Sprite (이미지)Mesh + Material
주의점2D와 3D 컴포넌트는 서로 충돌하지 않습니다. Collider2D와 Collider를 섞어 쓰면 충돌 이벤트가 영원히 안 옵니다. 신입이 반나절씩 잡아먹는 대표적 함정입니다.

렌더 파이프라인: 모바일이면 URP

유니티에는 렌더링 방식이 여러 개 있습니다. URP(Universal Render Pipeline)는 모바일을 포함한 다양한 기기에서 성능과 품질의 균형을 맞추도록 설계된 파이프라인으로, 모바일 신규 프로젝트의 사실상 표준입니다. HDRP는 고사양 PC·콘솔용이라 모바일에서는 선택지가 아닙니다. 프로젝트 생성 시 템플릿에서 URP를 고르면 설정이 자동으로 잡힙니다.

모바일이라서 신경 써야 하는 4가지

입력이 손가락이다

마우스 커서가 없으니 hover 상태가 없고, 손가락이 화면의 30%를 가립니다. 버튼은 최소 7~9mm(약 44pt) 이상, 중요한 정보는 손가락이 닿는 하단 중앙을 피해 배치합니다.

해상도와 비율이 제각각이다

16:9부터 20:9, 태블릿 4:3까지 대응해야 합니다. UI는 절대 좌표로 찍지 말고 앵커와 Canvas Scaler로 잡아야 합니다(3장 참고).

발열과 배터리가 성능 상한이다

PC와 달리 몇 분만 고부하로 돌려도 기기가 스스로 성능을 낮춥니다(thermal throttling). 벤치마크는 5분 이상 연속 플레이 후의 프레임으로 판단해야 합니다.

메모리 여유가 적다

저사양 기기는 앱이 쓸 수 있는 메모리가 1GB 안팎입니다. 4K 텍스처 몇 장으로 강제 종료(OOM)가 납니다. 텍스처 압축은 선택이 아니라 필수입니다.

프로젝트 폴더 구조와 버전 관리

유니티 프로젝트에서 실제로 관리해야 할 폴더는 Assets, Packages, ProjectSettings 세 개입니다. Library, Temp, Build는 자동 생성되는 캐시라 Git에 올리면 안 됩니다(용량이 수 GB에 이릅니다).

02 기본 문법

C# 문법 절반과
유니티 규칙 몇 개면
일단 게임은 돌아간다

C#을 전부 배우고 시작할 필요는 없습니다. 실제 게임 코드의 대부분은 변수, 조건문, 반복문, 클래스, 그리고 유니티가 정해둔 함수 이름 몇 개로 이루어져 있습니다. 여기서는 딱 그만큼만 다룹니다.

먼저 알아야 할 C# 기초

자주 쓰는 타입만 외우면 충분합니다
int   hp      = 100;        // 정수: 체력, 점수, 개수
float speed   = 5.5f;       // 실수: 속도, 시간 (반드시 뒤에 f)
bool  isAlive = true;       // 참/거짓: 상태 플래그
string playerName = "용사"; // 문자열

// 유니티 전용 타입
Vector3    pos   = new Vector3(1f, 2f, 0f); // 3차원 좌표
Vector2    dir   = new Vector2(1f, 0f);     // 2차원 (2D 게임에서 주로)
Color      red   = Color.red;               // 색상
GameObject go;                              // 게임 오브젝트 참조

// const(상수)와 readonly
const int MAX_LEVEL = 99;   // 절대 안 바뀌는 값은 상수로

// 접근 제한자: 외부에서 건드리면 안 되는 값은 private이 기본
public    int score;     // 어디서든 접근 가능 (남용 금지)
private   int comboCount; // 이 클래스 안에서만
protected int baseDamage; // 이 클래스 + 자식 클래스
float에 f를 빼먹으면?

float speed = 5.5;는 컴파일 에러입니다. C#에서 소수점 숫자의 기본 타입은 double이라 float에 바로 넣을 수 없습니다. 유니티는 float을 쓰므로 습관적으로 f를 붙이세요.

게임 로직의 90%는 이 두 가지
// 조건문
if (hp <= 0)
{
    Die();
}
else if (hp < 30)
{
    ShowLowHpWarning();
}

// && (그리고), || (또는), ! (반대)
if (isGrounded && !isDead)
{
    Jump();
}

// 삼항 연산자: 짧은 분기
string text = (hp > 0) ? "생존" : "사망";

// switch: 상태가 여러 개일 때 if보다 읽기 좋다
switch (state)
{
    case GameState.Ready:  ShowCountdown(); break;
    case GameState.Play:   RunGame();       break;
    case GameState.Over:   ShowResult();    break;
    default: break;

MonoBehaviour 생명주기 — 유니티만의 규칙

스크립트가 MonoBehaviour를 상속하면, 유니티가 정해진 이름의 함수를 정해진 시점에 알아서 호출합니다. 내가 호출하는 게 아니라 불려지는 구조입니다. 아래에서 버튼을 눌러 실제 호출 순서를 확인해 보세요.

생명주기 시뮬레이터 정지됨
Awake/Start는 한 번, Update는 매 프레임. 비활성화했다 켜면 OnEnable은 다시 불리지만 Start는 다시 불리지 않습니다.
함수호출 시점여기에 쓰면 좋은 것
Awake()오브젝트가 생성될 때 가장 먼저 (비활성 상태여도 실행)자기 자신의 컴포넌트 캐싱, 싱글톤 등록
OnEnable()활성화될 때마다이벤트 구독(+=)
Start()첫 Update 직전에 한 번 (모든 Awake 이후)다른 오브젝트 참조 가져오기, 초기 데이터 세팅
FixedUpdate()고정 간격(기본 0.02초)마다물리 연산, Rigidbody 제어
Update()매 프레임입력 감지, 일반 로직
LateUpdate()모든 Update가 끝난 뒤카메라 추적(캐릭터가 다 움직인 후 따라가야 떨림이 없음)
OnDisable()비활성화될 때이벤트 구독 해제(-=)
OnDestroy()파괴될 때정리 작업, 저장
Awake vs Start를 가르는 한 문장

"나 자신을 준비하는 일은 Awake, 남과 연결하는 일은 Start." 다른 오브젝트가 아직 Awake도 안 끝난 상태에서 그 값을 읽으려 하면 null이 나옵니다. 모든 Awake가 끝난 뒤에 Start가 돌기 때문에 이 순서를 지키면 초기화 순서 버그가 거의 사라집니다.

Time.deltaTime — 이 줄을 빼면 기기마다 게임이 달라진다

Update는 "1초에 60번"이 보장되지 않습니다. 고사양 기기는 120번, 저사양은 30번 돌 수 있습니다. 그래서 프레임당 얼마가 아니라 초당 얼마로 계산해야 합니다. Time.deltaTime은 이전 프레임에서 지금까지 걸린 시간(초)입니다.

deltaTime 있고 없고 비교 60 FPS 환경

코루틴 — 여러 프레임에 걸친 작업

"3초 기다렸다가 폭발", "체력바를 천천히 줄이기"처럼 시간이 흐르는 동작은 Update에 타이머 변수를 만드는 것보다 코루틴이 읽기 쉽습니다.

Assets/Scripts/Bomb.cs
using System.Collections;
using UnityEngine;

public class Bomb : MonoBehaviour
{
    void Start()
    {
        StartCoroutine(ExplodeAfter(3f)); // 코루틴은 StartCoroutine으로 시작
    }

    IEnumerator ExplodeAfter(float delay)
    {
        yield return new WaitForSeconds(delay);  // 여기서 3초 대기 (게임은 계속 진행)
        Explode();
    }

    // 여러 번 반복하는 패턴
    IEnumerator SpawnWave()
    {
        for (int i = 0; i < 10; i++)
        {
            Instantiate(enemyPrefab, RandomPos(), Quaternion.identity);
            yield return new WaitForSeconds(0.5f); // 0.5초 간격으로 10마리
        }
    }

    // 한 프레임씩 진행하며 값을 서서히 바꾸기 (페이드 아웃)
    IEnumerator FadeOut(float duration)
    {
        float t = 0f;
        while (t < duration)
        {
            t += Time.deltaTime;
            sr.color = new Color(1f, 1f, 1f, 1f - t / duration);
            yield return null;   // 다음 프레임까지 대기
        }
    }
}
yield 문의미
yield return null다음 프레임까지 기다린다
yield return new WaitForSeconds(2f)2초 대기 (timeScale 영향 받음)
yield return new WaitForSecondsRealtime(2f)2초 대기 (일시정지 중에도 흐름)
yield return new WaitUntil(() => isReady)조건이 참이 될 때까지 대기
yield return StartCoroutine(Other())다른 코루틴이 끝날 때까지 대기
코루틴의 함정 두 가지

1) 오브젝트가 비활성화되면 코루틴은 즉시 중단되고, 다시 켜도 이어지지 않습니다. 2) WaitForSeconds를 매번 new로 만들면 GC 쓰레기가 쌓이므로, 반복 호출되는 곳에서는 private readonly WaitForSeconds wait = new WaitForSeconds(0.5f);처럼 미리 만들어 재사용하세요.

이벤트 — 서로 모르는 채로 연결하기

"플레이어가 죽으면 UI도 바뀌고, 사운드도 나오고, 통계도 기록된다." 플레이어 스크립트가 이 셋을 직접 호출하면 서로 엉겨 붙습니다. "죽었다"고 방송만 하고, 관심 있는 쪽이 각자 듣는 구조로 만드세요.

    for (int i = 0; i < Input.touchCount; i++)

        Touch t = Input.GetTouch(i);

        switch (t.phase)

                break;

오브젝트 풀링 — 모바일 최적화의 첫 단추

총알을 쏠 때마다 Instantiate하고 Destroy하면, 메모리를 계속 새로 할당하고 버리게 됩니다. 버려진 메모리는 언젠가 GC(가비지 컬렉션)가 한꺼번에 정리하는데, 그 순간 프레임이 뚝 끊깁니다. 미리 만들어 두고 껐다 켜서 재사용하는 것이 풀링입니다.

Instantiate vs 풀링 연타해 보세요
신규 생성: 위쪽 0개 · 아래쪽 0개  |  누적 GC 대상 0개
위(빨강)는 매번 새로 생성 — 발사할수록 쓰레기가 쌓입니다. 아래(초록)는 풀에서 꺼내 쓰기 — 일정 개수 이후로 생성이 멈춥니다.
Assets/Scripts/Utils/ObjectPool.cs
using System.Collections.Generic;
using UnityEngine;

public class ObjectPool : MonoBehaviour
{
    [SerializeField] private GameObject prefab;
    [SerializeField] private int initialSize = 20;

    private readonly Queue<GameObject> pool = new Queue<GameObject>();

    void Awake()
    {
        for (int i = 0; i < initialSize; i++)
        {
            GameObject go = Instantiate(prefab, transform);
            go.SetActive(false);
            pool.Enqueue(go);
        }
    }

    public GameObject Get(Vector3 pos, Quaternion rot)
    {
        GameObject go = pool.Count > 0 ? pool.Dequeue() : Instantiate(prefab, transform);
        go.transform.SetPositionAndRotation(pos, rot);
        go.SetActive(true);
        return go;
    }

    public void Return(GameObject go)
    {
        go.SetActive(false);
        pool.Enqueue(go);
    }
}

// 사용하는 쪽: Destroy 대신 Return을 호출한다
public class Bullet : MonoBehaviour
{
    private ObjectPool pool;
    public void Init(ObjectPool p) => pool = p;

    void OnBecameInvisible() => pool.Return(gameObject);  // 화면 밖으로 나가면 회수
}

모바일 프레임 저하는 대부분 드로우콜, 텍스처 메모리, GC 할당 세 가지에서 옵니다. 항목을 눌러 진행 상황을 체크해 보세요.

드로우콜 줄이기

그림 한 장을 그리라는 명령 하나가 드로우콜입니다. 스프라이트 아틀라스로 이미지를 한 장에 모으면 여러 오브젝트가 한 번에 그려집니다(배칭). UI도 Canvas를 자주 바뀌는 것과 고정된 것으로 분리하세요. 하나만 바뀌어도 그 Canvas 전체가 다시 그려집니다.

텍스처 압축은 ASTC

임포트 설정에서 압축을 안 하면 1024×1024 이미지 한 장이 4MB를 먹습니다. ASTC는 현대 안드로이드·iOS 기기에서 널리 지원되는 표준 압축 포맷입니다. Max Size도 실제 표시 크기에 맞춰 낮추세요.

GC 할당을 만들지 않기

매 프레임 new, 문자열 더하기("점수:" + score), foreach의 일부 패턴이 쓰레기를 만듭니다. UI 텍스트는 값이 바뀔 때만 갱신하고, 문자열은 StringBuilder나 미리 만든 배열을 쓰세요.

Update 개수 줄이기

오브젝트 500개가 각자 Update를 가지면 호출 비용만으로 부담입니다. 매니저 하나가 리스트를 돌며 처리하는 구조로 바꾸거나, 실제로 필요 없는 Update는 빈 채로 두지 말고 삭제하세요(빈 Update도 호출 비용이 있습니다).

조명은 미리 굽기

3D라면 실시간 조명과 실시간 그림자는 모바일에서 매우 비쌉니다. 정적인 물체는 라이트맵을 베이크하고, 그림자는 단순한 블롭 텍스처로 대체하는 것이 일반적입니다.

해상도 스케일 조절

저사양 기기에서는 Screen.SetResolution이나 URP의 Render Scale을 0.7~0.8로 낮추면 화질 손해 대비 프레임 이득이 큽니다. 픽셀 수는 제곱으로 줄어듭니다.

using UnityEngine;
using UnityEngine.EventSystems;

[RequireComponent(typeof(Rigidbody2D))]
public class PlayerJump : MonoBehaviour
{
    [SerializeField] private float jumpPower = 9f;
    [SerializeField] private LayerMask groundLayer;
    [SerializeField] private Transform groundCheck;

    private Rigidbody2D rb;
    private Animator anim;
    private bool isGrounded;

    void Awake()
    {
        rb = GetComponent<Rigidbody2D>();
        anim = GetComponent<Animator>();
    }

    void Update()
    {
        // 발밑으로 짧은 원을 쏘아 바닥 판정
        isGrounded = Physics2D.OverlapCircle(groundCheck.position, 0.12f, groundLayer);
        anim.SetBool("IsGrounded", isGrounded);

        if (IsTapped() && isGrounded) Jump();
    }

    private bool IsTapped()
    {
        if (!Input.GetMouseButtonDown(0)) return false;
        if (EventSystem.current.IsPointerOverGameObject()) return false; // UI 위면 무시
        return true;
    }

    private void Jump()
    {
        rb.velocity = new Vector2(rb.velocity.x, 0f);
        rb.AddForce(Vector2.up * jumpPower, ForceMode2D.Impulse);
        anim.SetTrigger("Jump");
        SoundManager.Instance.PlaySfx(jumpClip);
    }

    void OnTriggerEnter2D(Collider2D other)
    {
        if (other.CompareTag("Obstacle")) GameManager.Instance.GameOver();
    }
}

using System.Collections;
using UnityEngine;

public class ObstacleSpawner : MonoBehaviour
{
    [SerializeField] private ObjectPool pool;
    [SerializeField] private float minDelay = 0.9f;
    [SerializeField] private float maxDelay = 1.8f;
    [SerializeField] private Transform spawnPoint;

    private Coroutine routine;

    void OnEnable()  => routine = StartCoroutine(SpawnLoop());
    void OnDisable() { if (routine != null) StopCoroutine(routine); }

    private IEnumerator SpawnLoop()
    {
        while (true)
        {
            yield return new WaitForSeconds(Random.Range(minDelay, maxDelay));

            if (GameManager.Instance.IsPlaying)
                pool.Get(spawnPoint.position, Quaternion.identity);
        }
    }
}

using UnityEngine;
using TMPro;

public class GameManager : MonoBehaviour
{
    public static GameManager Instance { get; private set; }

    [SerializeField] private TMP_Text scoreText;
    [SerializeField] private GameObject gameOverPanel;

    public bool IsPlaying { get; private set; }
    private float score;

    {
        if (Instance != null && Instance != this) { Destroy(gameObject); return; }
        Instance = this;
    }

    void Start()
    {
        IsPlaying = true;
        Time.timeScale = 1f;
    }

    void Update()
    {
        if (!IsPlaying) return;

        score += Time.deltaTime * 10f;
        scoreText.text = ((int)score).ToString();   // 매 프레임 갱신은 비용, 실제로는 정수가 바뀔 때만
    }

    public void GameOver()
    {
        if (!IsPlaying) return;
        IsPlaying = false;
        Time.timeScale = 0f;                        // 게임 정지

        int best = PlayerPrefs.GetInt("HighScore", 0);
        if ((int)score > best) PlayerPrefs.SetInt("HighScore", (int)score);

        gameOverPanel.SetActive(true);
    }

    public void Restart()
    {
        Time.timeScale = 1f;                        // 재시작 전 반드시 복구
        UnityEngine.SceneManagement.SceneManager.LoadScene("MainGame");
    }
}

반드시 실기기에 연결해서 측정하세요. PC는 폰보다 수십 배 빠르기 때문에 에디터 프로파일링 결과는 참고치도 되지 않습니다. Development Build + Autoconnect Profiler를 켜고 빌드하면 자동으로 연결됩니다.

전체적으로 프레임이 낮음CPU Usage의 Rendering / Scripts 비율드로우콜 과다, Update 로직 과부하

시간이 지날수록 느려짐Memory Profiler의 총 사용량 추이이벤트 해제 누락, 풀 회수 누락(메모리 누수)
